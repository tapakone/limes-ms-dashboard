null : Number((v + 2 * sigma).toFixed(2)))); const lower = mid.map(v => (v == null ? null : Number((v - 2 * sigma).toFixed(2)))); // --- Simple risk score (demo) --- const dayRisk = 3.5; const twoHrRisk = 3.1; const oneHrRisk = 4.2; function riskLabel(score) { if (score >= 4.0) return "HIGH RISK"; if (score >= 3.0) return "WATCH"; return "BUY"; } document.getElementById("status").textContent = `D Risk: ${dayRisk.toFixed(1)}/5 | 2H Risk: ${twoHrRisk.toFixed(1)}/5 | 1H Risk: ${oneHrRisk.toFixed(1)}/5`; // --- Draw chart --- const ctx = document.getElementById("chart").getContext("2d"); new Chart(ctx, { type: "line", data: { labels, datasets: [ { label: "Price", data: price, borderWidth: 2, tension: 0.25, }, { label: "Phase / Mid (MA3)", data: mid, borderWidth: 2, borderDash: [6, 6], tension: 0.25, }, { label: "Upper (+2σ)", data: upper, borderWidth: 1, borderDash: [3, 3], tension: 0.25, }, { label: "Lower (-2σ)", data: lower, borderWidth: 1, borderDash: [3, 3], tension: 0.25, }, ], }, options: { responsive: true, plugins: { legend: { labels: { color: "#ddd" }, }, title: { display: false, }, }, scales: { x: { ticks: { color: "#aaa" }, grid: { color: "rgba(255,255,255,0.08)" }, }, y: { ticks: { color: "#aaa" }, grid: { color: "rgba(255,255,255,0.08)" }, }, }, }, });
